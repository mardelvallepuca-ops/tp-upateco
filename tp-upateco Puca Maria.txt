import React, { useRef, useState } from 'react';
import emailjs from '@emailjs/browser';

const ContactForm = () => {
  const form = useRef();
  const [status, setStatus] = useState('');

  const sendEmail = (e) => {
    e.preventDefault();
    setStatus('enviando');

    emailjs
      .sendForm(
        'TU_SERVICE_ID',   // 🔴 Pegá acá tu Service ID de EmailJS
        'TU_TEMPLATE_ID',  // 🔴 Pegá acá tu Template ID de EmailJS
        form.current,
        'TU_PUBLIC_KEY'    // 🔴 Pegá acá tu Public Key de EmailJS
      )
      .then(() => {
        setStatus('ok');
        form.current.reset();
        setTimeout(() => setStatus(''), 4000);
      })
      .catch((error) => {
        console.error(error);
        setStatus('error');
      });
  };

  return (
    <form ref={form} onSubmit={sendEmail} className="contact-form">
      <div className="form-group">
        <label htmlFor="user_name">Nombre</label>
        <input
          id="user_name"
          type="text"
          name="user_name"
          placeholder="Maria Puca"
          required
        />
      </div>

      <div className="form-group">
        <label htmlFor="user_email">Correo electrónico</label>
        <input
          id="user_email"
          type="email"
          name="user_email"
          placeholder="tucorreo@ejemplo.com"
          required
        />
      </div>

      <div className="form-group">
        <label htmlFor="message">Mensaje</label>
        <textarea
          id="message"
          name="message"
          rows="5"
          placeholder="Escribe tu mensaje aquí..."
          required
        />
      </div>

      <button type="submit" disabled={status === 'enviando'}>
        {status === 'enviando' ? 'Enviando...' : 'Enviar Mensaje'}
      </button>

      {status === 'ok' && (
        <p className="msg ok">✅ ¡Mensaje enviado con éxito!</p>
      )}
      {status === 'error' && (
        <p className="msg error">❌ Hubo un error. Intenta nuevamente.</p>
      )}
    </form>
  );
};

export default ContactForm;
